src/app/api/chat/route.ts Streaming Edge Support 
import { GoogleGenerativeAI } from '@google/generative-ai';
import { NextRequest, NextResponse } from 'next/server';

// Validate API Key exists on server launch
const apiKey = process.env.GEMINI_API_KEY;
if (!apiKey) {
  console.error('[CRITICAL] Missing GEMINI_API_KEY environment variable.');
}

const genAI = new GoogleGenerativeAI(apiKey || '');

export async function POST(req: NextRequest) {
  try {
    if (!apiKey) {
      return NextResponse.json(
        { error: 'API key configuration error on server.' },
        { status: 500 }
      );
    }

    const { messages, model = 'gemini-1.5-flash' } = await req.json();

    if (!messages || !Array.isArray(messages) || messages.length === 0) {
      return NextResponse.json(
        { error: 'Invalid payload: Array of messages is required.' },
        { status: 400 }
      );
    }

    // Format conversation history into Gemini multi-turn format
    const formattedHistory = messages.slice(0, -1).map((m: any) => ({
      role: m.role === 'assistant' ? 'model' : 'user',
      parts: [{ text: m.content }],
    }));

    const latestPrompt = messages[messages.length - 1].content;
    const geminiModel = genAI.getGenerativeModel({ model });

    // Initiate multi-turn conversational session
    const chat = geminiModel.startChat({
      history: formattedHistory,
      generationConfig: {
        maxOutputTokens: 2048,
        temperature: 0.7,
      },
    });

    // Stream the generated chunks to client
    const result = await chat.sendMessageStream(latestPrompt);

    const encoder = new TextEncoder();
    const stream = new ReadableStream({
      async start(controller) {
        try {
          for await (const chunk of result.stream) {
            const chunkText = chunk.text();
            controller.enqueue(encoder.encode(chunkText));
          }
          controller.close();
        } catch (err) {
          controller.error(err);
        }
      },
    });

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/plain; charset=utf-8',
        'Transfer-Encoding': 'chunked',
        'Cache-Control': 'no-cache, no-transform',
      },
    });
  } catch (error: any) {
    // Handle Rate limits (429) & Auth key defects
    const status = error?.status === 429 ? 429 : 500;
    const message = error?.status === 429
      ? 'Gemini API rate limit exceeded. Please retry in a few seconds.'
      : (error?.message || 'Internal Generation Error');

    return NextResponse.json({ error: message }, { status });
  }
}
src/hooks/useChatStorage.ts Full CRUD Store 
'use client';

import { useState, useEffect, useCallback } from 'react';

export interface ChatMessage {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  timestamp: number;
}

export interface ChatSession {
  id: string;
  title: string;
  createdAt: number;
  updatedAt: number;
  messages: ChatMessage[];
}

const STORAGE_KEY = 'gemini_chat_sessions_v1';
const ACTIVE_SESSION_KEY = 'gemini_active_session_id';

export function useChatStorage() {
  const [sessions, setSessions] = useState<ChatSession[]>([]);
  const [activeId, setActiveId] = useState<string | null>(null);
  const [isLoaded, setIsLoaded] = useState<boolean>(false);

  // READ: Hydrate from LocalStorage safely on mount
  useEffect(() => {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      const parsed: ChatSession[] = raw ? JSON.parse(raw) : [];
      setSessions(parsed);

      const savedActive = localStorage.getItem(ACTIVE_SESSION_KEY);
      if (savedActive && parsed.some(s => s.id === savedActive)) {
        setActiveId(savedActive);
      } else if (parsed.length > 0) {
        setActiveId(parsed[0].id);
      }
    } catch (e) {
      console.error('Failed to parse chat storage:', e);
    } finally {
      setIsLoaded(true);
    }
  }, []);

  // Internal synchronizer
  const persist = useCallback((updated: ChatSession[]) => {
    setSessions(updated);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
  }, []);

  // CREATE: Spin up new conversational container
  const createSession = useCallback((customTitle = 'New Conversation') => {
    const newId = `session_` + Date.now().toString(36);
    const fresh: ChatSession = {
      id: newId,
      title: customTitle,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      messages: [],
    };
    persist([fresh, ...sessions]);
    setActiveId(newId);
    localStorage.setItem(ACTIVE_SESSION_KEY, newId);
    return fresh;
  }, [sessions, persist]);

  // UPDATE: Append message to currently selected conversation
  const appendMessage = useCallback((sessionId: string, msg: Omit<ChatMessage, 'id' | 'timestamp'>) => {
    const enriched: ChatMessage = {
      ...msg,
      id: `msg_` + Math.random().toString(36).substring(2, 9),
      timestamp: Date.now(),
    };

    const next = sessions.map(s => {
      if (s.id !== sessionId) return s;
      // Auto-title conversation on first human query
      const updatedTitle = s.messages.length === 0 && msg.role === 'user'
        ? msg.content.slice(0, 36) + (msg.content.length > 36 ? '...' : '')
        : s.title;

      return {
        ...s,
        title: updatedTitle,
        updatedAt: Date.now(),
        messages: [...s.messages, enriched],
      };
    });
    persist(next);
  }, [sessions, persist]);

  // UPDATE: Rename title explicitly
  const renameSession = useCallback((sessionId: string, title: string) => {
    persist(sessions.map(s => (s.id === sessionId ? { ...s, title, updatedAt: Date.now() } : s)));
  }, [sessions, persist]);

  // DELETE: Remove single session
  const deleteSession = useCallback((sessionId: string) => {
    const next = sessions.filter(s => s.id !== sessionId);
    persist(next);
    if (activeId === sessionId) {
      const fallback = next.length > 0 ? next[0].id : null;
      setActiveId(fallback);
      if (fallback) localStorage.setItem(ACTIVE_SESSION_KEY, fallback);
      else localStorage.removeItem(ACTIVE_SESSION_KEY);
    }
  }, [sessions, activeId, persist]);

  // DELETE: Nuclear purge of all history
  const clearAll = useCallback(() => {
    localStorage.removeItem(STORAGE_KEY);
    localStorage.removeItem(ACTIVE_SESSION_KEY);
    setSessions([]);
    setActiveId(null);
  }, []);

  const currentSession = sessions.find(s => s.id === activeId) || null;

  return {
    sessions,
    currentSession,
    activeId,
    setActiveId,
    isLoaded,
    createSession,
    appendMessage,
    renameSession,
    deleteSession,
    clearAll,
  };
}
src/lib/gemini.ts Client Stream Handler 
// Front-end streaming consumer bridge
export async function streamChatCompletion({
  messages,
  model = 'gemini-1.5-flash',
  onChunk,
}: {
  messages: { role: 'user' | 'assistant'; content: string }[];
  model?: string;
  onChunk: (chunkText: string) => void;
}) {
  const response = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ messages, model }),
  });

  if (!response.ok) {
    const errJson = await response.json().catch(() => ({}));
    throw new Error(errJson.error || `HTTP Error: ${response.status}`);
  }

  const reader = response.body?.getReader();
  if (!reader) throw new Error('No readable stream body returned');

  const decoder = new TextDecoder();
  let fullAccumulator = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value, { stream: true });
    fullAccumulator += text;
    onChunk(text);
  }

  return fullAccumulator;
}
